/* ---------- Process Creation ---------- */

void process_creation()
{
    pid_t pid;

    printf("\n========================================\n");
    printf("        PROCESS CREATION\n");
    printf("========================================\n");

    pid = fork();

    if (pid < 0)
    {
        perror("fork");
        return;
    }

    if (pid == 0)
    {
        printf("[CHILD PROCESS]\n");
        printf("Child PID        : %d\n", getpid());
        printf("Parent PID       : %d\n", getppid());

        printf("Child process completed.\n");
        exit(0);
    }
    else
    {
        printf("[PARENT PROCESS]\n");
        printf("Parent PID       : %d\n", getpid());
        printf("Created Child PID: %d\n", pid);

        wait(NULL);

        printf("Child process finished.\n");
    }
}

/* ---------- File I/O ---------- */

void file_operations()
{
    int fd;
    char buffer[200];

    const char *data =
        "OSSP Linux Debugging System\n"
        "File I/O test successful.\n";

    printf("\n========================================\n");
    printf("             FILE I/O\n");
    printf("========================================\n");

    /* Create and open file */
    fd = open("ossp_test.txt",
              O_CREAT | O_WRONLY | O_TRUNC,
              0644);

    if (fd < 0)
    {
        perror("open");
        return;
    }

    printf("File opened successfully.\n");

    /* Write data */
    if (write(fd, data, strlen(data)) < 0)
    {
        perror("write");
        close(fd);
        return;
    }

    printf("Data written successfully.\n");

    close(fd);

    printf("File closed.\n");

    /* Reopen for reading */
    fd = open("ossp_test.txt", O_RDONLY);

    if (fd < 0)
    {
        perror("open");
        return;
    }

    int bytes = read(fd, buffer, sizeof(buffer) - 1);

    if (bytes < 0)
    {
        perror("read");
        close(fd);
        return;
    }

    buffer[bytes] = '\0';

    printf("\nData read from file:\n");
    printf("----------------------------------------\n");
    printf("%s", buffer);
    printf("----------------------------------------\n");

    close(fd);

    printf("File operation completed successfully.\n");
}
